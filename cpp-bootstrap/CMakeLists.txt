add_library(devbox_setup src/setup.cpp)
target_include_directories(devbox_setup PUBLIC include)
target_link_libraries(devbox_setup PUBLIC devbox_core)
add_executable(devbox-setup src/main.cpp)
target_link_libraries(devbox-setup PRIVATE devbox_setup)
if(MSVC)
  target_compile_options(devbox_setup PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(devbox-setup PRIVATE /utf-8)
else()
  target_compile_options(devbox_setup PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(DEVBOX_BUILD_TESTS)
  add_executable(devbox-setup-tests tests/setup_tests.cpp)
  target_link_libraries(devbox-setup-tests PRIVATE devbox_setup)
  if(MSVC)
    target_compile_options(devbox-setup-tests PRIVATE /utf-8)
  endif()
  add_test(NAME setup COMMAND devbox-setup-tests $<TARGET_FILE:devbox-mcp>)
  set_tests_properties(setup PROPERTIES TIMEOUT 45)
endif()
