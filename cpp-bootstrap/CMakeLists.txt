add_library(devbox_setup src/setup.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DEVBOX_SETUP_VERSION LIMIT_COUNT 1)
if(NOT DEVBOX_SETUP_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "The C++ installer VERSION must be a semantic version")
endif()
target_compile_definitions(devbox_setup PUBLIC DEVBOX_SETUP_VERSION="${DEVBOX_SETUP_VERSION}")
target_include_directories(devbox_setup PUBLIC include)
target_link_libraries(devbox_setup PUBLIC devbox_core)
add_executable(devbox-setup src/main.cpp)
target_link_libraries(devbox-setup PRIVATE devbox_setup)
add_library(devbox_bundle src/bundle.cpp)
target_include_directories(devbox_bundle PUBLIC include)
target_link_libraries(devbox_bundle PUBLIC devbox_core)
add_executable(devbox-bundle-pack src/bundle_pack_main.cpp)
target_link_libraries(devbox-bundle-pack PRIVATE devbox_bundle)
set(DEVBOX_OFFLINE_MANIFEST "" CACHE FILEPATH "Compiled-in offline payload identity (generated by devbox-bundle-pack)")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(DEVBOX_OFFLINE_MANIFEST)
  file(READ "${DEVBOX_OFFLINE_MANIFEST}" offline_manifest HEX)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DEVBOX_OFFLINE_MANIFEST}")
else()
  set(offline_manifest "7b7d")
endif()
string(REGEX REPLACE "(..)" "0x\\1," offline_manifest_bytes "${offline_manifest}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/offline_manifest.hpp.tmp"
  "inline constexpr unsigned char embedded_offline_manifest[] = {${offline_manifest_bytes}0};\n")
file(COPY_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/offline_manifest.hpp.tmp"
  "${CMAKE_CURRENT_BINARY_DIR}/generated/offline_manifest.hpp" ONLY_IF_DIFFERENT)
add_executable(devbox-offline EXCLUDE_FROM_ALL src/offline_main.cpp)
target_include_directories(devbox-offline PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(devbox-offline PRIVATE devbox_bundle)
foreach(target IN ITEMS devbox_bundle devbox-bundle-pack devbox-offline)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endforeach()
if(MSVC)
  target_compile_options(devbox_setup PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(devbox-setup PRIVATE /utf-8)
else()
  target_compile_options(devbox_setup PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(DEVBOX_BUILD_TESTS)
  add_executable(devbox-bundle-tests tests/bundle_tests.cpp)
  target_link_libraries(devbox-bundle-tests PRIVATE devbox_bundle)
  if(MSVC)
    target_compile_options(devbox-bundle-tests PRIVATE /utf-8)
  endif()
  add_test(NAME offline-bundle COMMAND devbox-bundle-tests)
  set_tests_properties(offline-bundle PROPERTIES TIMEOUT 45)
  add_executable(devbox-setup-tests tests/setup_tests.cpp)
  target_link_libraries(devbox-setup-tests PRIVATE devbox_setup)
  if(MSVC)
    target_compile_options(devbox-setup-tests PRIVATE /utf-8)
  endif()
  add_test(NAME setup COMMAND devbox-setup-tests $<TARGET_FILE:devbox-mcp>)
  set_tests_properties(setup PROPERTIES TIMEOUT 45)
endif()
