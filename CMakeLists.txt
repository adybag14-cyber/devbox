cmake_minimum_required(VERSION 3.24)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
project(devbox_cpp VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS "19.43")
    message(FATAL_ERROR "Native C++23 builds require MSVC 19.43 or newer")
  endif()
  # CMake otherwise maps C++23 to /std:c++latest, which also enables later
  # experimental language changes. Use the compiler's explicit C++23 mode.
  set(CMAKE_CXX23_STANDARD_COMPILE_OPTION "-std:c++23preview")
  set(CMAKE_CXX23_EXTENSION_COMPILE_OPTION "-std:c++23preview")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(NOT DEFINED CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  endif()
endif()
option(DEVBOX_BUILD_TESTS "Build native C++ verification" ON)
option(DEVBOX_BUILD_BENCHMARKS "Build native component timing probes" OFF)
option(DEVBOX_BUILD_TUI "Build existing native setup UI" ON)
option(DEVBOX_SANITIZERS "Instrument native code with address and undefined-behavior sanitizers" OFF)
option(DEVBOX_MSVC_ASAN "Instrument Windows native verification with MSVC AddressSanitizer" OFF)
option(DEVBOX_THREAD_SANITIZER "Instrument a separate Linux thread-sanitizer qualification" OFF)
option(DEVBOX_STATIC_ANALYSIS "Run MSVC native static analysis during compilation" OFF)
option(DEVBOX_FUZZING "Build bounded Clang libFuzzer parser targets" OFF)
if(DEVBOX_THREAD_SANITIZER AND (DEVBOX_SANITIZERS OR DEVBOX_MSVC_ASAN))
  message(FATAL_ERROR "ThreadSanitizer must run separately from AddressSanitizer")
endif()
if(DEVBOX_MSVC_ASAN)
  if(NOT MSVC)
    message(FATAL_ERROR "DEVBOX_MSVC_ASAN requires MSVC")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
  add_compile_options(/fsanitize=address /Zi)
  add_compile_definitions(DEVBOX_SANITIZERS=1)
endif()
if(DEVBOX_THREAD_SANITIZER)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR MSVC)
    message(FATAL_ERROR "ThreadSanitizer qualification is supported on native Linux")
  endif()
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
  add_compile_definitions(DEVBOX_SANITIZERS=1)
endif()
if(DEVBOX_STATIC_ANALYSIS)
  if(NOT MSVC)
    message(FATAL_ERROR "DEVBOX_STATIC_ANALYSIS requires MSVC")
  endif()
  add_compile_options(/analyze /analyze:external-)
endif()
option(DEVBOX_FORCE_FORK_EXEC "Exercise the Android API 21 process launch path on other POSIX hosts" OFF)
if(DEVBOX_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use Clang or GCC for the address/undefined-behavior sanitizer gate")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  add_compile_definitions(DEVBOX_SANITIZERS=1)
endif()
include(CTest)
add_subdirectory(cpp-mcp)
add_subdirectory(cpp-bootstrap)
if(DEVBOX_BUILD_TUI)
  add_subdirectory(setup-tui)
endif()
install(TARGETS devbox-mcp devbox-setup RUNTIME DESTINATION bin)
install(FILES cpp-mcp/libdeflate.LICENSE.txt DESTINATION share/devbox)
install(FILES cpp-mcp/lexbor.LICENSE.txt DESTINATION share/devbox)
install(FILES cpp-mcp/pugixml.LICENSE.txt DESTINATION share/devbox)
